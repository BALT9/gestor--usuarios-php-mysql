2. crear tabla en mysql

CREATE TABLE Usuarios (
	id int AUTO_INCREMENT PRIMARY KEY,
    ci VARCHAR(10) NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(100) NOT NULL UNIQUE,
    fecha DATE
);
